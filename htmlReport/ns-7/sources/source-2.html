 package com.suplementoslucho.tienda.service.impl;
 
 import constant.com.lucho.tienda.CartConstants;
 import constant.com.lucho.tienda.ErrorMessageConstants;
 import dto.com.lucho.tienda.ProductOperationRequest;
 import exception.com.lucho.tienda.BadRequestException;
 import exception.com.lucho.tienda.ResourceNotFoundException;
 import model.com.lucho.tienda.Cart;
 import model.com.lucho.tienda.Product;
 import model.com.lucho.tienda.User;
 import repository.com.lucho.tienda.CartRepository;
 import repository.com.lucho.tienda.ProductRepository;
 import repository.com.lucho.tienda.UserRepository;
 import service.com.lucho.tienda.CartService;
 import strategy.com.lucho.tienda.DiscountStrategy;
 import lombok.RequiredArgsConstructor;
 import lombok.extern.slf4j.Slf4j;
 import org.springframework.orm.ObjectOptimisticLockingFailureException;
 import org.springframework.scheduling.annotation.Async;
 import org.springframework.stereotype.Service;
 import org.springframework.transaction.annotation.Transactional;
 
 import java.math.BigDecimal;
 import java.util.ArrayList;
 import java.util.List;
 
 @Slf4j
 @Service
 @RequiredArgsConstructor
 public class CartServiceImpl implements CartService, CartConstants, ErrorMessageConstants {
 
     private final CartRepository cartRepository;
     private final UserRepository userRepository;
     private final ProductRepository productRepository;
     private final DiscountStrategy discountStrategy;
 
     @Override
     @Transactional
     public Cart createCart(Long userId) {
         if (userId == null) {
             throw new BadRequestException(USER_ID_REQUIRED);
         }
 
         User user = userRepository.findById(userId)
                 .orElseThrow(() -> new ResourceNotFoundException(String.format(USER_NOT_FOUND, userId)));
 
         Cart cart = new Cart();
         cart.setUser(user);
         cart.setProducts(new ArrayList<>());
         return cartRepository.save(cart);
     }
 
     @Override
     @Transactional
     public Cart addProduct(ProductOperationRequest request) {
         validateProductRequest(request);
         Cart cart = getCartWithProducts(request.cartId());
         Product product = getProduct(request.productCode());
 
         cart.getProducts().add(product);
         return cartRepository.save(cart);
     }
 
     @Override
     @Transactional
     public Cart removeProduct(ProductOperationRequest request) {
         validateProductRequest(request);
         Cart cart = getCartWithProducts(request.cartId());
         Product product = getProduct(request.productCode());
 
         cart.getProducts().remove(product);
         return cartRepository.save(cart);
     }
 
     @Override
     @Transactional(readOnly = true)
     public List<Product> getCartProducts(Long cartId) {
         if (cartId == null) {
             throw new BadRequestException(CART_ID_REQUIRED);
         }
         return getCartWithProducts(cartId).getProducts();
     }
 
     @Override
     @Transactional(readOnly = true)
     public List<Cart> getUserCarts(Long userId) {
         if (userId == null) {
             throw new BadRequestException(USER_ID_REQUIRED);
         }
         return cartRepository.findByUserId(userId);
     }
 
     @Override
     @Async
     public void processOrderAsync(Long cartId) {
         if (cartId == null) {
             return;
         }
 
         Cart cart = cartRepository.findCartById(cartId)
                 .orElseThrow(() -> new ResourceNotFoundException(String.format(CART_NOT_FOUND, cartId)));
 
         if (STATUS_PROCESSED.equalsIgnoreCase(cart.getStatus())) {
             log.info("El carrito ID {} ya fue procesado previamente.", cartId);
             return;
         }
 
         BigDecimal totalAmount = calculateCartTotal(cart);
 
         try {
             Thread.sleep(ASYNC_SIMULATION_DELAY_MS);
         } catch (InterruptedException e) {
             Thread.currentThread().interrupt();
             log.error("Procesamiento asíncrono interrumpido para el carrito ID {}", cartId, e);
         }
 
         updateCartStatusSafely(cartId, totalAmount);
     }
 
     @Override
     @Transactional
     public void updateCartStatusSafely(Long cartId, BigDecimal totalAmount) {
         if (cartId == null) {
             return;
         }
 
         try {
             Cart cart = cartRepository.findCartById(cartId)
                     .orElseThrow(() -> new ResourceNotFoundException(String.format(CART_NOT_FOUND, cartId)));
 
             if (!STATUS_PROCESSED.equalsIgnoreCase(cart.getStatus())) {
                 cart.setStatus(STATUS_PROCESSED);
                 cart.setTotalAmount(totalAmount != null ? totalAmount : BigDecimal.ZERO);
                 cartRepository.save(cart);
                 log.info("Orden procesada exitosamente para carrito {}. Total persistido: ${}", cartId, totalAmount);
             }
         } catch (ObjectOptimisticLockingFailureException e) {
             log.warn("Conflicto de concurrencia evitado por @Version en carrito ID: {}", cartId);
         }
     }
 
     @Override
     @Transactional(readOnly = true)
     public Cart getCartById(Long cartId) {
         if (cartId == null) {
             throw new BadRequestException(CART_ID_REQUIRED);
         }
         return getCartWithProducts(cartId);
     }
 
     @Override
     public BigDecimal calculateCartTotal(Cart cart) {
         BigDecimal subtotal = cart.getProducts().stream()
                 .map(Product::getPrice)
                 .reduce(BigDecimal.ZERO, BigDecimal::add);
 
         BigDecimal discount = discountStrategy.calculateDiscount(cart);
         return subtotal.subtract(discount);
     }
 
     // --- Auxiliares ---
 
     private Cart getCartWithProducts(Long cartId) {
         return cartRepository.findCartById(cartId)
                 .orElseThrow(() -> new ResourceNotFoundException(String.format(CART_NOT_FOUND, cartId)));
     }
 
     private Product getProduct(String code) {
         return productRepository.findByCode(code)
                 .orElseThrow(() -> new ResourceNotFoundException(String.format(PRODUCT_NOT_FOUND, code)));
     }
 
     private void validateProductRequest(ProductOperationRequest request) {
         if (request == null || request.cartId() == null || request.productCode() == null || request.productCode().isBlank()) {
             throw new BadRequestException(INVALID_PRODUCT_REQUEST);
         }
     }
 }
