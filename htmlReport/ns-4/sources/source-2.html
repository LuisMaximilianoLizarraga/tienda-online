 package com.suplementoslucho.tienda.dto;
 
 import model.com.lucho.tienda.Cart;
 import java.math.BigDecimal;
 import java.util.List;
 import java.util.stream.Collectors;
 
 public record CartResponse(
         Long id,
         String status,
         BigDecimal totalAmount,
         Long userId,
         List<ProductResponse> products
 ) {
     public static CartResponse fromEntity(Cart cart) {
         if (cart == null) return null;
 
         return new CartResponse(
                 cart.getId(),
                 cart.getStatus(),
                 cart.getTotalAmount(),
                 cart.getUser() != null ? cart.getUser().getId() : null,
                 cart.getProducts().stream()
                         .map(ProductResponse::fromEntity) // Usamos el factory de ProductResponse
                         .collect(Collectors.toList())
         );
     }
     public static List<CartResponse> fromEntityList(List<Cart> carts) {
         if (carts == null) return List.of(); // Protección contra nulos
 
         return carts.stream()
                 .map(CartResponse::fromEntity) // Reutiliza el mapeo individual
                 .toList();
     }
 }
