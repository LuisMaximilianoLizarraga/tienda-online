 package com.suplementoslucho.tienda.controller;
 
 import com.suplementoslucho.tienda.dto.*;
 import model.com.lucho.tienda.Cart;
 import model.com.lucho.tienda.Product;
 import service.com.lucho.tienda.CartService;
 import io.swagger.v3.oas.annotations.Operation;
 import io.swagger.v3.oas.annotations.responses.ApiResponse;
 import io.swagger.v3.oas.annotations.tags.Tag;
 import lombok.RequiredArgsConstructor;
 import org.springframework.beans.factory.annotation.Autowired;
 import org.springframework.http.HttpStatus;
 import org.springframework.http.ResponseEntity;
 import org.springframework.validation.annotation.Validated;
 import org.springframework.web.bind.annotation.*;
 
 import java.util.List;
 
 @RestController
 @RequestMapping("/api/carts")
 @RequiredArgsConstructor
 @Tag(name = "Cart Controller", description = "Endpoints para la gestión de carritos y procesamiento de órdenes")
 public class CartController {
     @Autowired
     private final CartService cartService;
 
     @Operation(summary = "Crear un nuevo carrito", description = "Crea un carrito de compras asociado a un ID de usuario.")
     @ApiResponse(responseCode = "201", description = "Carrito creado exitosamente")
     @PostMapping
     public ResponseEntity<CartResponse> createCart(@RequestBody @Validated CreateCartRequest request) {
         Cart cart = cartService.createCart(request.userId());
         return ResponseEntity.status(HttpStatus.CREATED).body(CartResponse.fromEntity(cart));
     }
 
     @Operation(summary = "Agregar un producto al carrito")
     @ApiResponse(responseCode = "200", description = "Producto agregado correctamente")
     @PostMapping("/products")
     public ResponseEntity<CartResponse> addProduct(@RequestBody @Validated ProductOperationRequest request) {
         Cart cart = cartService.addProduct(request);
         return ResponseEntity.ok(CartResponse.fromEntity(cart));
     }
 
     @Operation(summary = "Eliminar un producto del carrito")
     @ApiResponse(responseCode = "200", description = "Producto removido correctamente")
     @DeleteMapping("/products")
     public ResponseEntity<CartResponse> removeProduct(@RequestBody @Validated ProductOperationRequest request) {
         Cart cart = cartService.removeProduct(request);
         return ResponseEntity.ok(CartResponse.fromEntity(cart));
     }
 
     @Operation(summary = "Obtener los productos de un carrito")
     @ApiResponse(responseCode = "200", description = "Lista de productos devuelta exitosamente")
     @GetMapping("/{cartId}/products")
     public ResponseEntity<List<ProductResponse>> getCartProducts(@PathVariable("cartId") Long cartId) {
         List<Product> products = cartService.getCartProducts(cartId);
         return ResponseEntity.ok(ProductResponse.fromEntityList(products));
     }
 
     @Operation(summary = "Listar carritos por usuario")
     @ApiResponse(responseCode = "200", description = "Lista de carritos del usuario")
     @GetMapping("/user/{userId}")
     public ResponseEntity<List<CartResponse>> getUserCarts(@PathVariable("userId") Long userId) {
         List<Cart> carts = cartService.getUserCarts(userId);
         return ResponseEntity.ok(CartResponse.fromEntityList(carts));
     }
 
     @Operation(summary = "Procesar la orden de forma asíncrona")
     @ApiResponse(responseCode = "202", description = "Solicitud aceptada y en proceso de ejecución")
     @PostMapping("/{cartId}/process")
     public ResponseEntity<AsyncOrderResponse> processOrder(@PathVariable("cartId") Long cartId) {
         cartService.processOrderAsync(cartId);
         AsyncOrderResponse response = new AsyncOrderResponse("Estamos procesando su orden");
         return ResponseEntity.status(HttpStatus.ACCEPTED).body(response);
     }
 
     @Operation(summary = "Obtener carrito por id")
     @ApiResponse(responseCode = "200", description = "Retorna el carrito")
     @GetMapping("/{id}")
     public ResponseEntity<CartResponse> getCart(@PathVariable("id") Long id) {
         Cart cart = cartService.getCartById(id);
         return ResponseEntity.ok(CartResponse.fromEntity(cart));
     }
 }
