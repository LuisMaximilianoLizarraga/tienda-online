 package com.suplementoslucho.tienda.model;
 
 import jakarta.persistence.*;
 import lombok.*;
 import java.math.BigDecimal;
 import java.util.List;
 
 @Entity
 @Table(name = "carritos") // Nombre físico en la BD
 @Getter
 @Setter
 @NoArgsConstructor
 @AllArgsConstructor
 @Builder
 public class Cart {
 
     @Id
     @GeneratedValue(strategy = GenerationType.IDENTITY)
     private Long id;
 
     @ManyToOne(fetch = FetchType.LAZY)
     @JoinColumn(name = "usuario_id") // Clave foránea física en español
     private User user;
 
     @ManyToMany
     @JoinTable(
             name = "carrito_productos", // Tabla intermedia física
             joinColumns = @JoinColumn(name = "carrito_id"),
             inverseJoinColumns = @JoinColumn(name = "producto_id")
     )
     private List<Product> products;
 
     private String status;
 
     private BigDecimal totalAmount;
 
     @Version
     private Long version;
 }
